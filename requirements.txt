# Runtime dependencies for the CricketVision backend.
#
# These are the modules the tracking, pitch, shot and HTTP API code actually
# imports. Install with:
#
#     pip install -r requirements.txt
#
# The upstream project pins exact versions; this manifest uses compatible
# ranges so the module also installs against a newer environment. The exact
# versions the upstream project was verified against are listed next to each
# entry.
#
# torch/torchvision: for CPU-only installs, use the PyTorch CPU index first and
# drop the minimum pins:
#     pip install "torch>=2.5" "torchvision>=0.20" --index-url https://download.pytorch.org/whl/cpu

# ── Vision and maths ──────────────────────────────────────────────────────────
# upstream pins: numpy==2.1.3, opencv-python==4.10.0.84
numpy>=2.1,<3
opencv-python>=4.10,<6

# ── Detection ─────────────────────────────────────────────────────────────────
# ultralytics provides YOLO11 (`ultralytics.YOLO`), the ball detector's model
# runtime. torch is its inference backend and is imported lazily, so the module
# can be imported (and its pure-maths parts tested) without a GPU.
# upstream pins: ultralytics==8.4.162, torch==2.5.1, torchvision==0.20.1
ultralytics>=8.4,<9
torch>=2.5
torchvision>=0.20

# ── HTTP API ──────────────────────────────────────────────────────────────────
# FastAPI serves the six-route schema 2.1 contract; Uvicorn is the server.
# python-multipart is required for the multipart upload routes, and httpx backs
# the FastAPI test client.
fastapi>=0.115,<1
uvicorn[standard]>=0.30,<1
python-multipart>=0.0.9,<1
httpx>=0.27,<1

# ── Configuration ─────────────────────────────────────────────────────────────
# Reads an optional `.env` from the project root so a model path can be
# overridden without editing code. The import is optional at runtime.
python-dotenv>=1.0,<2

# ── Development and verification ──────────────────────────────────────────────
# upstream pin: pytest==9.1.1
pytest>=8,<10
# Linter. The rules and the line length are pinned in ruff.toml.
ruff>=0.6
