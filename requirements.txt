# Runtime and verification dependencies for the CricketVision backend.
#
# Install:
#
#     python -m pip install -r requirements.txt
#
# PyTorch is the only GPU-sensitive dependency. For a CPU-only installation,
# install the CPU wheels first (they satisfy the pinned versions below):
#
#     pip install --index-url https://download.pytorch.org/whl/cpu \
#         torch==2.13.0 torchvision==0.28.0
#     pip install -r requirements.txt
#
# For an NVIDIA GPU, use the CUDA 12.6 wheel index instead:
#
#     pip install --index-url https://download.pytorch.org/whl/cu126 \
#         torch==2.13.0 torchvision==0.28.0
#     pip install -r requirements.txt
#
# The Docker build selects the same wheel index with TORCH_INDEX_URL. See
# README.md and compose.gpu.yaml. Nothing downloads model weights automatically.

# ── Vision and maths ─────────────────────────────────────────────────────────
numpy==2.5.2
opencv-python==5.0.0.93

# ── Detection ────────────────────────────────────────────────────────────────
# ultralytics provides YOLO11 (`ultralytics.YOLO`), the ball and pitch detector
# runtime. torch is its inference backend. Install torch/torchvision from the
# CPU or CUDA index above so the correct build is selected.
ultralytics==8.4.135
torch==2.13.0
torchvision==0.28.0

# ── HTTP API ─────────────────────────────────────────────────────────────────
# FastAPI serves the schema 2.1 contract. python-multipart is required for the
# upload routes; httpx backs the FastAPI test client.
fastapi==0.141.1
uvicorn[standard]==0.52.0
python-multipart==0.0.32
httpx==0.28.1

# ── Configuration ────────────────────────────────────────────────────────────
# Reads an optional `.env` from the project root so model paths and limits can
# be overridden without editing code.
python-dotenv==1.2.3

# ── Development and verification ─────────────────────────────────────────────
pytest==9.1.1
ruff==0.17.1
